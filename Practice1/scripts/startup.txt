// Stage 5: Additional Commands

// Copy file
cp home/user/welcome.txt home/user/welcome_copy.txt
ls home/user
head home/user/welcome_copy.txt

// Copy directory
cp -r home/user/documents home/user/documents_backup
ls home/user
ls home/user/documents_backup
head home/user/documents_backup/info.txt

// Copy errors
cp missing.txt copy.txt
cp home/user/documents backup

// Remove file
rm home/user/welcome_copy.txt
ls home/user

// Remove directory
rm -r home/user/documents_backup
ls home/user

// Remove errors
rm missing.txt
rm home/user/documents

// Invalid arguments
rm
